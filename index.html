        let bluetoothDevice = null;

        async function connectBluetooth() {

            try {

                UI.btStatus.innerText = "Szukam trenażera...";

                // Szukamy serwisu Cycling Power (0x1818)

                bluetoothDevice = await navigator.bluetooth.requestDevice({

                    filters: [{ services: ['cycling_power'] }]

                });


                bluetoothDevice.addEventListener('gattserverdisconnected', onDisconnected);

                

                UI.btStatus.innerText = "Łączenie...";

                const server = await bluetoothDevice.gatt.connect();

                const service = await server.getPrimaryService('cycling_power');

                powerCharacteristic = await service.getCharacteristic('cycling_power_measurement');

                

                await powerCharacteristic.startNotifications();

                powerCharacteristic.addEventListener('characteristicvaluechanged', handlePowerMeasurement);

                

                lastBtPacketTime = Date.now();

                

                UI.btStatus.innerText = "Podłączono! 🚲";

                UI.btStatus.classList.remove('text-red-500');

                UI.btStatus.classList.add('text-green-500');

                UI.btBtn.style.display = 'none';

                UI.kbdBtn.style.display = 'none';

            } catch (error) {

                // Środowisko iframe często blokuje Web Bluetooth ze względów bezpieczeństwa.

                // Usunięto console.error(), aby Canvas nie przechwytywał i nie wyświetlał okna błędu.

                

                // Obsługa błędu uprawnień w środowisku podglądu

                if (error.name === 'SecurityError' || (error.message && error.message.includes('permissions policy'))) {

                    UI.btStatus.innerText = "Brak uprawnień BT w tym podglądzie. Użyj klawiatury!";

                } else if (error.name === 'NotFoundError') {

                    UI.btStatus.innerText = "Nie wybrano urządzenia lub anulowano.";

                } else {

                    UI.btStatus.innerText = "Błąd połączenia: " + error.message;

                }

                

                UI.btStatus.classList.remove('text-green-500');

                UI.btStatus.classList.add('text-red-500');

            }

        }


        function onDisconnected() {

            UI.btStatus.innerText = "Utracono sygnał! Wznawianie...";

            UI.btStatus.classList.remove('text-green-500');

            UI.btStatus.classList.add('text-yellow-500');

            

            // Próba automatycznego wznowienia połączenia (Re-connect)

            setTimeout(reconnectBluetooth, 2000);

        }


        async function reconnectBluetooth() {

            if (!bluetoothDevice) return;

            try {

                const server = await bluetoothDevice.gatt.connect();

                const service = await server.getPrimaryService('cycling_power');

                powerCharacteristic = await service.getCharacteristic('cycling_power_measurement');

                await powerCharacteristic.startNotifications();

                

                UI.btStatus.innerText = "Podłączono ponownie! 🚲";

                UI.btStatus.classList.remove('text-yellow-500');

                UI.btStatus.classList.add('text-green-500');

                lastBtPacketTime = Date.now();

            } catch (e) {

                UI.btStatus.innerText = "Odłączono (błąd wznowienia).";

                UI.btStatus.classList.remove('text-yellow-500');

                UI.btStatus.classList.add('text-red-500');

                UI.btBtn.style.display = 'inline-block';

            }

        }


        function handlePowerMeasurement(event) {

            lastBtPacketTime = Date.now();

            const value = event.target.value;

            // Standard Cycling Power Measurement (0x2A63)

            // Flagi to pierwsze 2 bajty

            // Moc (Waty) to kolejne 2 bajty (sint16)

            let power = value.getInt16(2, true); 

            myPower = power;

            UI.hudPower.innerText = myPower + " W";

        }

                // Spadek mocy jest płynny w logice gry

        function listenToRoom(roomId) {

            // Nasłuchiwanie zmian w pokoju (np. start gry)

            unsubscribeRoom = onSnapshot(getRoomRef(roomId), (docSnapshot) => {

                if(docSnapshot.exists()) {

                    const data = docSnapshot.data();

                    if(data.status === 'racing' && !isRacing) {

                        startGameLoop();

                    }

                    if(data.hostId === myPlayerId) {

                        UI.startRaceBtn.style.display = 'inline-block';

                    }

                }

            }, (error) => console.error("Room listener error:", error));


            // Nasłuchiwanie graczy

            unsubscribePlayers = onSnapshot(getPlayersRef(roomId), (snapshot) => {

                playersMap.clear();

                UI.playersList.innerHTML = '';

                

                snapshot.forEach(doc => {

                    const data = doc.data();

                    playersMap.set(doc.id, data);

                    

                    const li = document.createElement('div');

                    li.className = 'mb-2 p-2 border-2 border-gray-600 bg-gray-800 flex justify-between items-center';

                    li.innerHTML = `

                        <span>

                            <span style="color:${data.color}">●</span> 

                            ${data.name} ${doc.id === myPlayerId ? '(Ty)' : ''}

                        </span>

                        <span class="text-xs text-gray-400">Laps: ${data.laps}/${TOTAL_LAPS}</span>

                    `;

                    UI.playersList.appendChild(li);

                });

            }, (error) => console.error("Players listener error:", error));

        }

        function getTrackPosition(distancePct, laneIndex, totalLanes) {

            // Tor to prostokąt z zaokrąglonymi bokami (stadion)

            const cw = canvas.width;

            const ch = canvas.height;

            

            // Środek i wymiary bazowe

            const cx = cw / 2;

            const cy = ch / 2;

            const straightLen = cw * 0.4; // długość prostej

            const baseRadius = ch * 0.3;  // promień wirażu

            

            // Odstępy pasów

            const laneWidth = (ch * 0.15) / Math.max(totalLanes, 1);

            const r = baseRadius + (laneIndex * laneWidth) - ((totalLanes * laneWidth)/2);

            

            // Punkty kontrolne

            const leftCenterX = cx - straightLen/2;

            const rightCenterX = cx + straightLen/2;

            

            // Obwód częściowy

            const curveLen = Math.PI * r;

            const totalLen = 2 * straightLen + 2 * curveLen;

            

            // Pozycja lokalna 0..totalLen, przesunięta o połowę prostej, 

            // aby wartość 0.0 (start/meta) idealnie pokrywała się z wizualną linią

            let d = ((distancePct % 1.0) * totalLen + straightLen/2) % totalLen;

            if (d < 0) d += totalLen; // zabezpieczenie

            

            let x, y, angle;


            if (d < straightLen) {

                // Dolna prosta (od lewej do prawej)

                x = leftCenterX + d;

                y = cy + r;

                angle = 0;

            } else if (d < straightLen + curveLen) {

                // Prawy wiraż (od dołu do góry)

                let arcD = d - straightLen;

                let a = Math.PI/2 - (arcD / curveLen) * Math.PI; // od 90 do -90 stopni

                x = rightCenterX + Math.cos(a) * r;

                y = cy + Math.sin(a) * r;

                angle = a - Math.PI/2;

            } else if (d < 2*straightLen + curveLen) {

                // Górna prosta (od prawej do lewej)

                let strD = d - (straightLen + curveLen);

                x = rightCenterX - strD;

                y = cy - r;

                angle = Math.PI;

            } else {

                // Lewy wiraż (od góry do dołu)

                let arcD = d - (2*straightLen + curveLen);

                let a = -Math.PI/2 - (arcD / curveLen) * Math.PI; // od -90 do -270 stopni

                x = leftCenterX + Math.cos(a) * r;

                y = cy + Math.sin(a) * r;

                angle = a - Math.PI/2;

            }


            return {x, y, angle};

        }


        function gameLoop(time) {

            if(!isRacing) return;

            

            const dt = (time - lastTime) / 1000; // sekundy

            lastTime = time;


            // 1. Fizyka lokalna (Tylko ja)

            // Utrata mocy, jeśli używamy klawiatury (symulacja oporu)

            if(isUsingKeyboard && myPower > 0) {

                myPower -= 80 * dt;

                if(myPower < 0) myPower = 0;

                UI.hudPower.innerText = Math.round(myPower) + " W";

            } else if (!isUsingKeyboard && bluetoothDevice && bluetoothDevice.gatt.connected) {

                // Trenażer Watchdog: jeśli pakiety BT utkną na >2.5s, zerujemy moc

                if (Date.now() - lastBtPacketTime > 2500 && myPower > 0) {

                    myPower = 0;

                    UI.hudPower.innerText = "0 W";

                }

            }


            // Prędkość na podstawie mocy (proste mapowanie: 300W = 1 okrążenie / 10s)

            // Prędkość w % okrążenia na sekundę

            let speed = (myPower / 300) * 0.1; 

            

            if(myLaps < TOTAL_LAPS) {

                myDistance += speed * dt;

                if(myDistance >= 1.0) {

                    myDistance -= 1.0;

                    myLaps++;

                    UI.hudLaps.innerText = myLaps + "/" + TOTAL_LAPS;

                    

                    if(myLaps >= TOTAL_LAPS) {

                        myDistance = 0; // Zatrzymaj na starcie

                        myPower = 0; // Odcięcie gazu po przekroczeniu mety

                        UI.hudPower.innerText = "0 W";

                        UI.hudPos.innerText = "META!";

                    }

                }

            }


            // 2. Synchronizacja z Firebase (Throttle do 10Hz by uniknąć limitów)

            dbUpdateTimer += dt;

            if(dbUpdateTimer > 0.1 && myLaps <= TOTAL_LAPS) {

                dbUpdateTimer = 0;

                // Asynchroniczny update - nie blokujemy rysowania

                updateDoc(getMyPlayerRef(currentRoomId), {

                    distance: myDistance,

                    laps: myLaps,

                    power: Math.round(myPower)

                }).catch(e => console.warn("DB Update skipped", e));

            }


            // 3. Rysowanie

            drawTrack();

            drawPlayers();


            requestAnimationFrame(gameLoop);

        }


        function drawTrack() {

            ctx.fillStyle = '#4a5d23'; // Trawa w środku

            ctx.fillRect(0, 0, canvas.width, canvas.height);


            const cx = canvas.width/2;

            const cy = canvas.height/2;

            const straightLen = canvas.width * 0.4;

            const baseRadius = canvas.height * 0.3;

            const trackWidth = canvas.height * 0.25;


            // Zewnętrzny obrys (banda)

            ctx.lineWidth = trackWidth;

            ctx.strokeStyle = '#3f2a14'; // Żużel

            ctx.lineCap = 'butt';


            ctx.beginPath();

            ctx.arc(cx - straightLen/2, cy, baseRadius, Math.PI/2, Math.PI*1.5);

            ctx.lineTo(cx + straightLen/2, cy - baseRadius);

            ctx.arc(cx + straightLen/2, cy, baseRadius, -Math.PI/2, Math.PI/2);

            ctx.lineTo(cx - straightLen/2, cy + baseRadius);

            ctx.stroke();


            // Wewnętrzna linia murawy

            ctx.lineWidth = 4;

            ctx.strokeStyle = '#fff';

            ctx.beginPath();

            ctx.arc(cx - straightLen/2, cy, baseRadius - trackWidth/2, Math.PI/2, Math.PI*1.5);

            ctx.lineTo(cx + straightLen/2, cy - (baseRadius - trackWidth/2));

            ctx.arc(cx + straightLen/2, cy, baseRadius - trackWidth/2, -Math.PI/2, Math.PI/2);

            ctx.lineTo(cx - straightLen/2, cy + (baseRadius - trackWidth/2));

            ctx.stroke();


            // Linia Startu/Mety - Szachownica

            const startX = cx;

            const startY = cy + baseRadius - trackWidth/2;

            

            ctx.fillStyle = '#fff';

            ctx.fillRect(startX - 8, startY, 16, trackWidth);

            

            ctx.fillStyle = '#000';

            for (let i = 0; i < trackWidth; i += 8) {

                ctx.fillRect(startX - 8, startY + i, 8, 4);

                if (i + 4 < trackWidth) {

                    ctx.fillRect(startX, startY + i + 4, 8, 4);

                }

            }


        }


        function drawPlayers() {

            let index = 0;

            const totalPlayers = Math.max(playersMap.size, 1);

            

            // Sortowanie graczy wg progresu (Laps + distance)

            let sortedPlayers = Array.from(playersMap.values()).sort((a,b) => {

                let progA = a.laps + a.distance;

                let progB = b.laps + b.distance;

                return progB - progA;

            });


            // Aktualizacja HUD Pozycji (dla mnie)

            let myRank = sortedPlayers.findIndex(p => p.name === myName) + 1;

            if(myRank > 0 && myLaps < TOTAL_LAPS) UI.hudPos.innerText = "Poz: " + myRank;


            // Rysuj

            sortedPlayers.forEach((p, idx) => {

                // Zachowaj stały tor dla danego gracza bazując na indeksie by nie skakali

                // Można by hashować ID gracza, ale uprośćmy - tor losowy bazując na kolorze

                let laneSeed = parseInt(p.color.substring(1,3), 16) % totalPlayers;


                let pos = getTrackPosition(p.distance, laneSeed, totalPlayers);

                

                ctx.save();

                ctx.translate(pos.x, pos.y);

                ctx.rotate(pos.angle);


                // Motocykl (Prosty kwadracik)

                ctx.fillStyle = p.color;

                ctx.fillRect(-15, -8, 30, 16);

                

                // Koła

                ctx.fillStyle = '#000';

                ctx.fillRect(-12, -10, 8, 4);

                ctx.fillRect(-12, 6, 8, 4);

                ctx.fillRect(10, -10, 8, 4);

                ctx.fillRect(10, 6, 8, 4);


                // Nazwa gracza nad

                ctx.rotate(-pos.angle); // Zawsze poziomo

                ctx.fillStyle = '#fff';

                ctx.font = '10px "Press Start 2P"';

                ctx.textAlign = 'center';

                ctx.fillText(p.name, 0, -15);

                

                // Info o końcu

                if(p.laps >= TOTAL_LAPS) {

                    ctx.fillStyle = '#ff0';

                    ctx.fillText("META", 0, 20);

                }


                ctx.restore();

            });


            // Ekran końca gry dla gracza

            if (myLaps >= TOTAL_LAPS) {

                ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';

                ctx.fillRect(0, 0, canvas.width, canvas.height);

                ctx.fillStyle = '#fff';

                ctx.font = '20px "Press Start 2P"';

                ctx.textAlign = 'center';

                ctx.fillText("WYŚCIG ZAKOŃCZONY", canvas.width/2, canvas.height/2);

                ctx.font = '10px "Press Start 2P"';

                ctx.fillStyle = '#f59e0b';

                ctx.fillText("Oczekuj na pozostałych zawodników...", canvas.width/2, canvas.height/2 + 30);

            }

        }
